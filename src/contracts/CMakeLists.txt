# Build generated C01 C++ messages and descriptor-driven boundary validation.
# Integration: protoc and libprotobuf must both be 34.1; no network/model initialization.
# Generated sources are build artifacts. Wire tests are separate from model performance gates.
cmake_minimum_required(VERSION 3.20)
project(regulagraph_contracts LANGUAGES CXX)
find_package(Protobuf 34.1.0 EXACT CONFIG REQUIRED)
find_program(REGULAGRAPH_PROTOC protoc REQUIRED)
execute_process(COMMAND "${REGULAGRAPH_PROTOC}" --version OUTPUT_VARIABLE PROTOC_VERSION OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT PROTOC_VERSION STREQUAL "libprotoc 34.1")
  message(FATAL_ERROR "Expected protoc 34.1, got ${PROTOC_VERSION}")
endif()
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(PROTO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/proto")
set(GEN_ROOT "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GEN_ROOT}")
set(PROTO_FILES common documents graph evidence answers jobs inference)
set(PROTO_INPUTS)
set(PROTO_OUTPUTS)
foreach(NAME IN LISTS PROTO_FILES)
  list(APPEND PROTO_INPUTS "${PROTO_ROOT}/regulagraph/v1/${NAME}.proto")
  list(APPEND PROTO_OUTPUTS "${GEN_ROOT}/regulagraph/v1/${NAME}.pb.cc" "${GEN_ROOT}/regulagraph/v1/${NAME}.pb.h")
endforeach()
list(APPEND PROTO_INPUTS "${REPO_ROOT}/evaluation/datasets/evaluation.proto")
list(APPEND PROTO_OUTPUTS "${GEN_ROOT}/evaluation.pb.cc" "${GEN_ROOT}/evaluation.pb.h")
get_target_property(PROTOBUF_INCLUDE protobuf::libprotobuf INTERFACE_INCLUDE_DIRECTORIES)
add_custom_command(OUTPUT ${PROTO_OUTPUTS}
 COMMAND "${REGULAGRAPH_PROTOC}" "--proto_path=${PROTO_ROOT}" "--proto_path=${REPO_ROOT}/evaluation/datasets"
 "--proto_path=${PROTOBUF_INCLUDE}" "--cpp_out=${GEN_ROOT}" ${PROTO_INPUTS}
 DEPENDS ${PROTO_INPUTS} VERBATIM)
add_library(regulagraph_wire STATIC ${PROTO_OUTPUTS} wire_validation.cpp)
target_compile_features(regulagraph_wire PUBLIC cxx_std_17)
target_include_directories(regulagraph_wire PUBLIC "${GEN_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(regulagraph_wire PUBLIC protobuf::libprotobuf)
if(MSVC)
 target_compile_options(regulagraph_wire PRIVATE /bigobj)
endif()
option(REGULAGRAPH_WIRE_TESTS "Build cross-language wire fixture executable" ON)
if(REGULAGRAPH_WIRE_TESTS)
 add_executable(regulagraph_wire_probe "${REPO_ROOT}/tests/integration/wire_cpp.cpp")
 target_link_libraries(regulagraph_wire_probe PRIVATE regulagraph_wire)
endif()
